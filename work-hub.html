Firestore 보안 규칙 (업무톡용 추가분)

Firebase 콘솔 → Firestore Database → 규칙 에 아래 블록을 기존 규칙 안에 추가하세요. 로그인한 사용자만 읽고 쓰며, 본인 글만 지울 수 있습니다.

match /hubTalk/{msg} {
  allow read:   if request.auth != null;
  allow create: if request.auth != null
                && request.resource.data.uid == request.auth.uid;
  allow update: if request.auth != null;   // 고정/해제
  allow delete: if request.auth != null
                && resource.data.uid == request.auth.uid;
}
match /hubChannels/{ch} { allow read, write: if request.auth != null; }
match /hubPrograms/{d}  { allow read, write: if request.auth != null; }
match /hubSessions/{d}  { allow read, write: if request.auth != null; }
match /hubItems/{d}     { allow read, write: if request.auth != null; }
match /hubLedger/{d}    { allow read, write: if request.auth != null; }
match /hubConfig/{d}    { allow read, write: if request.auth != null; }
